      -- CIA data request dataset: pollock_610_620
      -- 
      -- Grid:      Hexagonal grid, 20 km (data.hex_20km)
      -- Filters:   species_code IN ('POLL')
      -- Spatial:   include hauls overlapping NMFS reporting area 610 or 620
      -- Grouping:  grid cell, year
      -- 
      -- Confidentiality:
      --   - Each record aggregates catch from at least 3 distinct vessels.
      -- 
      -- Method: each haul's catch is apportioned across the grid cells its footprint
      -- overlaps, in proportion to the overlapping area. Area-only hauls use the full
      -- statistical area as their footprint.
      -- Overlaps covering less than 5% of a haul's area are dropped (except AREA hauls).
      -- Shares are renormalised to sum to 1 per haul.
      -- 
      -- Units:     catch_weight_mt = metric tons; catch_count = number of fish.
      -- CRS:       EPSG:3338 (NAD83 / Alaska Albers).
      -- Generated by ciarequests <version>
      WITH
      -- 1. Hauls matching the haul-level filters and spatial filters.
      --    A haul matches an area if their footprints share some area (touching
      --    boundaries does not count); matching hauls are kept or dropped whole.
      --    The checks are written "(...) IS TRUE" so PostgreSQL evaluates them per
      --    haul instead of as a join, whose row estimates would distort the plan.
      hauls_f AS (
          SELECT h.haul_id,
                 h.haul_polygon,
                 ST_Area(h.haul_polygon) AS haul_area,
                 h.vessel_id,
                 h.year,
                 h.haul_type
          FROM data.hauls h
          WHERE (EXISTS (  -- include hauls overlapping NMFS reporting area 610 or 620
                SELECT 1 FROM data.management_areas sa
                WHERE sa.source = 'NMFS_REPORTING_AREA'
                  AND sa.area_code::text IN ('610', '620')
                  AND ST_Intersects(sa.shape, h.haul_polygon)
                  AND NOT ST_Touches(sa.shape, h.haul_polygon))) IS TRUE
      ),
      -- 2. Overlay the grid: share of each haul's area falling in each cell.
      --    The intersection is computed only where a cell cuts through a haul:
      --    a haul inside one cell has share 1, and a cell inside a haul (common
      --    for area hauls) contributes the cell's area. Zero-area overlaps (cells
      --    merely touching a haul's edge) are dropped.
      overlay AS (
          SELECT * FROM (
              SELECT f.haul_id,
                     f.haul_type,
                     g.hexid AS grid_id,
                     CASE WHEN ST_CoveredBy(f.haul_polygon, g.shape) THEN 1.0
                          WHEN ST_CoveredBy(g.shape, f.haul_polygon) THEN ST_Area(g.shape) / f.haul_area
                          ELSE ST_Area(ST_Intersection(f.haul_polygon, g.shape)) / f.haul_area
                     END AS p_raw
              FROM hauls_f f
              JOIN data.hex_20km g ON ST_Intersects(f.haul_polygon, g.shape)
              OFFSET 0  -- computes p_raw once per pair (keeps the WHERE below from being
                        -- merged into this query, which would compute it twice)
          ) o
          WHERE p_raw > 0
      ),
      -- 3. Drop overlaps covering less than 5% of a haul's area, so slivers
      --    neither receive catch nor count the vessel toward that cell.
      --    Exempt: AREA hauls (their footprint is a whole statistical area).
      --    A haul with no overlap at or above the threshold keeps all of its overlaps.
      trimmed AS (
          SELECT haul_id, grid_id, p_raw
          FROM (SELECT o.*, MAX(p_raw) OVER (PARTITION BY haul_id) AS p_max FROM overlay o) o
          WHERE p_raw >= 0.05
             OR haul_type IN ('AREA')
             OR p_max < 0.05
      ),
      -- 4. Renormalise so each haul's shares sum to 1 (covers trimmed overlaps,
      --    hauls extending past the grid edge, and rounding).
      alloc AS (
          SELECT haul_id, grid_id,
                 p_raw / SUM(p_raw) OVER (PARTITION BY haul_id) AS p
          FROM trimmed
      ),
      -- 5. Join catch, apply catch-level filters, apportion catch by share
      apportioned AS (
          SELECT a.grid_id,
                 f.vessel_id,
                 f.year,
                 f.haul_type,
                 c.weight * a.p AS weight_mt,
                 c.count * a.p AS count_n
          FROM alloc a
          JOIN hauls_f f ON f.haul_id = a.haul_id
          JOIN data.haul_catch c ON c.haul_id = a.haul_id
          WHERE c.species_code IN ('POLL')
      ),
      -- 6. Aggregate by grid cell and grouping fields. Confidentiality rules:
      --    groups that fail any rule are not returned.
      agg AS (
          SELECT grid_id, year,
                 SUM(weight_mt) AS catch_weight_mt,
                 SUM(count_n)   AS catch_count
          FROM apportioned
          GROUP BY grid_id, year
          HAVING COUNT(DISTINCT vessel_id) >= 3  -- rule: min_vessels
      )
      SELECT a.grid_id, a.year,
             a.catch_weight_mt,
             a.catch_count,
             g.shape AS grid_shape
      FROM agg a
      JOIN data.hex_20km g ON g.hexid = a.grid_id
      ORDER BY a.grid_id, a.year;
